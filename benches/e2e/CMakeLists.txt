cmake_minimum_required(VERSION 4.0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rpc_pipeline_bench LANGUAGES CXX)
  enable_testing()
endif()
if(NOT TARGET benchmark::benchmark)
  find_package(benchmark CONFIG REQUIRED)
endif()
if(NOT TARGET task_gateway)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../examples/task_service"
    "${CMAKE_CURRENT_BINARY_DIR}/task-service-example")
endif()
add_executable(bench_rpc_pipeline bench_rpc_pipeline.cc)
target_link_libraries(bench_rpc_pipeline PRIVATE task_gateway benchmark::benchmark)
get_target_property(benchmark_includes benchmark::benchmark INTERFACE_INCLUDE_DIRECTORIES)
if(benchmark_includes)
  target_include_directories(bench_rpc_pipeline SYSTEM BEFORE PRIVATE ${benchmark_includes})
endif()

option(RPC_PIPELINE_BUILD_COMPONENT_BENCHMARKS "Benchmark canonical codecs and compression" ON)
if(RPC_PIPELINE_BUILD_COMPONENT_BENCHMARKS)
  if(NOT TARGET fmt::fmt)
    find_package(fmt CONFIG REQUIRED)
  endif()
  get_filename_component(repo_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  option(GRPCXX_ENABLE_PROTOBUF "Enable Protobuf codec" ON)
  option(GRPCXX_ENABLE_FORY "Enable Apache Fory codec" ON)
  option(GRPCXX_ENABLE_CAPNP "Enable CapnProto codec" ON)
  if(GRPCXX_ENABLE_CAPNP AND NOT CMAKE_C_COMPILER_LOADED)
    enable_language(C)
  endif()
  if(GRPCXX_ENABLE_FORY AND NOT TARGET fory::serialization)
    find_package(Fory CONFIG REQUIRED)
  endif()
  if(NOT TARGET grpcxx::codec)
    add_subdirectory("${repo_root}/src/grpcxx/codec" "${CMAKE_CURRENT_BINARY_DIR}/canonical-codec")
  endif()
  if(NOT TARGET grpcxx::compression)
    add_subdirectory("${repo_root}/src/grpcxx/compression" "${CMAKE_CURRENT_BINARY_DIR}/canonical-compression")
  endif()
  target_sources(bench_rpc_pipeline PRIVATE bench_component_roundtrips.cc)
  target_link_libraries(bench_rpc_pipeline PRIVATE grpcxx::compression)
  foreach(backend protobuf fory capnp)
    if(TARGET grpcxx::codec_${backend})
      target_link_libraries(bench_rpc_pipeline PRIVATE grpcxx::codec_${backend})
    endif()
  endforeach()
  target_link_libraries(bench_rpc_pipeline PRIVATE grpcxx::framing)
endif()
