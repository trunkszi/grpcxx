cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(grpcxx_reflection_benchmark LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 26)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  get_filename_component(GRPCXX_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  add_subdirectory("${GRPCXX_SOURCE_ROOT}/src/grpcxx/reflection" "${CMAKE_CURRENT_BINARY_DIR}/reflection")
endif()
find_package(benchmark CONFIG REQUIRED)
add_executable(bench_service_router bench_service_router.cc)
target_link_libraries(bench_service_router PRIVATE grpcxx::reflection benchmark::benchmark_main)
target_compile_options(bench_service_router PRIVATE -fno-exceptions -Wall -Wextra -Wswitch)
target_include_directories(bench_service_router PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
if(NOT TARGET grpcxx::core AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/grpcxx/core/status.cc")
  target_sources(bench_service_router PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/grpcxx/core/status.cc")
endif()
