cmake_minimum_required(VERSION 4.0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(task_service_e2e LANGUAGES CXX)
  enable_testing()
endif()
if(NOT TARGET GTest::gtest_main)
  find_package(GTest CONFIG REQUIRED)
endif()
if(NOT TARGET task_gateway)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../examples/task_service"
    "${CMAKE_CURRENT_BINARY_DIR}/task-service-example")
endif()
add_executable(task_service_e2e test_task_service_e2e.cc)
target_link_libraries(task_service_e2e PRIVATE task_gateway GTest::gtest_main)
target_compile_definitions(task_service_e2e PRIVATE GTEST_HAS_EXCEPTIONS=0)
add_test(NAME task_service_e2e COMMAND task_service_e2e)
set_tests_properties(task_service_e2e PROPERTIES LABELS "e2e;app;task_gateway;pipeline")
