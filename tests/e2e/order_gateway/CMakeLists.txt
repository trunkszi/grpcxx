cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(order_gateway_e2e LANGUAGES CXX)
  enable_testing()
endif()
if(NOT TARGET GTest::gtest_main)
  find_package(GTest CONFIG REQUIRED)
endif()
if(NOT TARGET order_gateway)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../examples/order_service"
    "${CMAKE_CURRENT_BINARY_DIR}/order-service-example")
endif()
add_executable(order_gateway_e2e test_order_gateway_e2e.cc)
target_link_libraries(order_gateway_e2e PRIVATE order_gateway GTest::gtest_main)
target_compile_definitions(order_gateway_e2e PRIVATE GTEST_HAS_EXCEPTIONS=0)
add_test(NAME order_gateway_e2e COMMAND order_gateway_e2e)
set_tests_properties(order_gateway_e2e PROPERTIES LABELS "e2e;app;order_gateway;pipeline")
