cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(grpcxx_h2_tests LANGUAGES CXX)
  enable_testing()
  set(CMAKE_CXX_STANDARD 26)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  get_filename_component(GRPCXX_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
  add_subdirectory("${GRPCXX_SOURCE_ROOT}/src/grpcxx/h2" "${CMAKE_CURRENT_BINARY_DIR}/h2")
endif()
find_package(GTest CONFIG REQUIRED)
add_executable(test_nghttp2_in_memory_unary test_nghttp2_in_memory_unary.cc)
target_link_libraries(test_nghttp2_in_memory_unary PRIVATE grpcxx::h2 GTest::gtest_main)
target_compile_options(test_nghttp2_in_memory_unary PRIVATE -fno-exceptions -Wall -Wextra -Wswitch)
add_test(NAME grpcxx.h2.in_memory_unary COMMAND test_nghttp2_in_memory_unary)
set_tests_properties(grpcxx.h2.in_memory_unary PROPERTIES LABELS "unit;library;h2;session")
target_include_directories(test_nghttp2_in_memory_unary PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
if(NOT TARGET grpcxx::core AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/grpcxx/core/status.cc")
  target_sources(test_nghttp2_in_memory_unary PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/grpcxx/core/status.cc")
endif()
