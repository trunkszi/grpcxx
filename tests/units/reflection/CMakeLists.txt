cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(grpcxx_reflection_tests LANGUAGES CXX)
  enable_testing()
  set(CMAKE_CXX_STANDARD 26)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  get_filename_component(GRPCXX_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
  add_subdirectory("${GRPCXX_SOURCE_ROOT}/src/grpcxx/reflection" "${CMAKE_CURRENT_BINARY_DIR}/reflection")
  add_subdirectory("${GRPCXX_SOURCE_ROOT}/src/grpcxx/h2" "${CMAKE_CURRENT_BINARY_DIR}/h2")
endif()
find_package(GTest CONFIG REQUIRED)
add_executable(test_cpp26_reflection_router test_cpp26_reflection_router.cc)
target_link_libraries(test_cpp26_reflection_router PRIVATE grpcxx::reflection grpcxx::h2 GTest::gtest_main)
target_compile_options(test_cpp26_reflection_router PRIVATE -fno-exceptions -Wall -Wextra -Wswitch)
add_test(NAME grpcxx.reflection.router COMMAND test_cpp26_reflection_router)
set_tests_properties(grpcxx.reflection.router PROPERTIES LABELS "unit;library;reflection;router")
target_include_directories(test_cpp26_reflection_router PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src")
if(NOT TARGET grpcxx::core AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/grpcxx/core/status.cc")
  target_sources(test_cpp26_reflection_router PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/grpcxx/core/status.cc")
endif()
if(NOT TARGET grpcxx::core AND NOT TARGET grpcxx::framing AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/grpcxx/framing/aligned_message_buf.cc")
  target_sources(test_cpp26_reflection_router PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/grpcxx/framing/aligned_message_buf.cc")
elseif(TARGET grpcxx::framing)
  target_link_libraries(test_cpp26_reflection_router PRIVATE grpcxx::framing)
endif()
