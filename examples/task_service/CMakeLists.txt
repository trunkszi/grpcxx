cmake_minimum_required(VERSION 4.0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(task_service LANGUAGES CXX)
  enable_testing()
endif()
if(NOT TARGET fmt::fmt)
  find_package(fmt CONFIG REQUIRED)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/canonical_dependencies.cmake")
task_service_load_canonical_dependencies()

if(NOT TARGET task_gateway)
  add_library(task_gateway STATIC task_gateway_service.cc task_transport.cc)
  target_include_directories(task_gateway PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(task_gateway PUBLIC grpcxx::core grpcxx::framing
    grpcxx::reflection grpcxx::h2 grpcxx::streaming)
  add_executable(task_service main.cc)
  target_link_libraries(task_service PRIVATE task_gateway fmt::fmt)
  add_test(NAME task_service_smoke COMMAND task_service)
  set_tests_properties(task_service_smoke PROPERTIES LABELS "e2e;app;task_service;smoke")
endif()

# Deferred registration sees optional suites added later by the integrating root.
function(task_service_register_existing_smokes)
  foreach(suite cpp26_reflection_unary streaming_showcase multi_codec_showcase tls_and_compression)
    if(TARGET ${suite} AND NOT TEST ${suite}_smoke)
      add_test(NAME ${suite}_smoke COMMAND $<TARGET_FILE:${suite}>)
      set_tests_properties(${suite}_smoke PROPERTIES LABELS "e2e;app;task_service;smoke")
    endif()
  endforeach()
endfunction()
get_property(task_service_smokes_scheduled GLOBAL PROPERTY TASK_SERVICE_SMOKES_SCHEDULED)
if(NOT task_service_smokes_scheduled)
  set_property(GLOBAL PROPERTY TASK_SERVICE_SMOKES_SCHEDULED TRUE)
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL task_service_register_existing_smokes)
endif()
