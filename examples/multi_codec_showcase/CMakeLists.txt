add_executable(multi_codec_showcase main.cc)
target_link_libraries(multi_codec_showcase PRIVATE grpcxx::reflection grpcxx::h2 fmt::fmt)

set(order_book_idl_dir "${CMAKE_CURRENT_SOURCE_DIR}/idl")
set(order_book_generated_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(multi_codec_showcase PRIVATE "${order_book_generated_dir}")

foreach(backend IN ITEMS proto fory capnp)
  set(generator_dependencies)
  if(backend STREQUAL "proto")
    if(NOT GRPCXX_ENABLE_PROTOBUF)
      continue()
    endif()
    find_package(Protobuf REQUIRED)
    set(extension proto)
    set(codec_target grpcxx::codec_protobuf)
    set(compiler_args --protoc "$<TARGET_FILE:protobuf::protoc>")
    set(binding_sources "${order_book_generated_dir}/proto/order_book.pb.cc")
    set(binding_headers "${order_book_generated_dir}/proto/order_book.pb.h")
  elseif(backend STREQUAL "fory")
    if(NOT GRPCXX_ENABLE_FORY)
      continue()
    endif()
    set(extension fdl)
    set(codec_target grpcxx::codec_fory)
    set(compiler_args)
    set(binding_sources)
    set(binding_headers "${order_book_generated_dir}/fory/cpp/examples_order_book_fory.h")
    list(APPEND generator_dependencies
      "${CMAKE_CURRENT_SOURCE_DIR}/../../src/grpcxx/tools/grpcxx_codegen/fdl_contract.py")
  else()
    if(NOT GRPCXX_ENABLE_CAPNP)
      continue()
    endif()
    find_package(CapnProto CONFIG REQUIRED)
    set(extension capnp)
    set(codec_target grpcxx::codec_capnp)
    target_link_libraries(multi_codec_showcase PRIVATE CapnProto::capnp-rpc)
    get_target_property(order_book_capnp_includes CapnProto::capnp INTERFACE_INCLUDE_DIRECTORIES)
    set(compiler_args)
    foreach(directory IN LISTS order_book_capnp_includes)
      list(APPEND compiler_args -I "${directory}")
    endforeach()
    set(binding_sources "${order_book_generated_dir}/capnp/order_book.capnp.c++")
    set(binding_headers "${order_book_generated_dir}/capnp/order_book.capnp.h")
  endif()

  set(output_dir "${order_book_generated_dir}/${backend}")
  set(schema "${order_book_idl_dir}/order_book.${extension}")
  set(contract "${output_dir}/order_book.grpcxx.h")
  add_custom_command(OUTPUT "${contract}" ${binding_sources} ${binding_headers}
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${output_dir}"
    COMMAND $<TARGET_FILE:grpcxx-codegen> --out "${output_dir}"
      ${compiler_args} -I "${order_book_idl_dir}" "${schema}"
    DEPENDS grpcxx-codegen protoc-gen-grpcxx "${schema}" ${generator_dependencies}
    VERBATIM COMMAND_EXPAND_LISTS)
  target_sources(multi_codec_showcase PRIVATE "${contract}" ${binding_sources} ${binding_headers})
  target_link_libraries(multi_codec_showcase PRIVATE ${codec_target})
endforeach()
