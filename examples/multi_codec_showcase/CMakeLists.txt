cmake_minimum_required(VERSION 4.0)
add_executable(multi_codec_showcase main.cc)
target_link_libraries(multi_codec_showcase PRIVATE grpcxx::reflection grpcxx::h2 fmt::fmt)

set(record_batch_idl_dir "${CMAKE_CURRENT_SOURCE_DIR}/idl")
set(record_batch_generated_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(multi_codec_showcase PRIVATE "${record_batch_generated_dir}")

foreach(backend IN ITEMS proto fory capnp)
  set(generator_dependencies)
  if(backend STREQUAL "proto")
    if(NOT GRPCXX_ENABLE_PROTOBUF)
      continue()
    endif()
    set(protobuf_MODULE_COMPATIBLE ON)
    find_package(Protobuf CONFIG QUIET)
    if(NOT Protobuf_FOUND)
      find_package(Protobuf REQUIRED)
    endif()
    set(extension proto)
    set(codec_target grpcxx::codec_protobuf)
    if(COMMAND grpcxx_host_tool)
      grpcxx_host_tool(_grpcxx_host_protoc "protobuf/protoc")
    endif()
    if(_grpcxx_host_protoc)
      set(compiler_args --protoc "${_grpcxx_host_protoc}")
    else()
      set(compiler_args --protoc "$<TARGET_FILE:protobuf::protoc>")
    endif()
    set(binding_sources "${record_batch_generated_dir}/proto/record_batch.pb.cc")
    set(binding_headers "${record_batch_generated_dir}/proto/record_batch.pb.h")
  elseif(backend STREQUAL "fory")
    if(NOT GRPCXX_ENABLE_FORY)
      continue()
    endif()
    set(extension fdl)
    set(codec_target grpcxx::codec_fory)
    set(compiler_args)
    set(binding_sources)
    set(binding_headers "${record_batch_generated_dir}/fory/cpp/examples_record_batch_fory.h")
    list(APPEND generator_dependencies
      "${CMAKE_CURRENT_SOURCE_DIR}/../../src/grpcxx/tools/grpcxx_codegen/fdl_contract.py")
  else()
    if(NOT GRPCXX_ENABLE_CAPNP)
      continue()
    endif()
    find_package(CapnProto CONFIG REQUIRED)
    set(extension capnp)
    set(codec_target grpcxx::codec_capnp)
    target_link_libraries(multi_codec_showcase PRIVATE CapnProto::capnp-rpc)
    # kj-async exports KJ_USE_FIBERS, which the kj headers reject under -fno-exceptions.
    target_compile_options(multi_codec_showcase PRIVATE -fexceptions)
    get_target_property(record_batch_capnp_includes CapnProto::capnp INTERFACE_INCLUDE_DIRECTORIES)
    set(compiler_args)
    foreach(directory IN LISTS record_batch_capnp_includes)
      list(APPEND compiler_args -I "${directory}")
    endforeach()
    set(binding_sources "${record_batch_generated_dir}/capnp/record_batch.capnp.c++")
    set(binding_headers "${record_batch_generated_dir}/capnp/record_batch.capnp.h")
  endif()

  set(output_dir "${record_batch_generated_dir}/${backend}")
  set(schema "${record_batch_idl_dir}/record_batch.${extension}")
  set(contract "${output_dir}/record_batch.grpcxx.h")
  add_custom_command(OUTPUT "${contract}" ${binding_sources} ${binding_headers}
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${output_dir}"
    COMMAND "${CMAKE_COMMAND}" -E env
      "GRPCXX_FDL_ADAPTER=${PROJECT_SOURCE_DIR}/src/grpcxx/tools/grpcxx_codegen/fdl_contract.py"
      $<TARGET_FILE:grpcxx-codegen> --out "${output_dir}"
      ${compiler_args} -I "${record_batch_idl_dir}" "${schema}"
    DEPENDS grpcxx-codegen protoc-gen-grpcxx "${schema}" ${generator_dependencies}
    VERBATIM COMMAND_EXPAND_LISTS)
  target_sources(multi_codec_showcase PRIVATE "${contract}" ${binding_sources} ${binding_headers})
  target_link_libraries(multi_codec_showcase PRIVATE ${codec_target})
endforeach()

if(GRPCXX_BUILD_TESTING OR BUILD_TESTING)

  set_tests_properties(integration.example.codec.multi_codec_showcase PROPERTIES
    LABELS "integration;example;codec;multi_codec" TIMEOUT 30)
