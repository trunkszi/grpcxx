get_filename_component(GRPCXX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.30)
  project(grpcxx_tls_and_compression LANGUAGES CXX)
  set(CMAKE_CXX26_STANDARD_COMPILE_OPTION "-std=c++26")
  set(CMAKE_CXX26_EXTENSION_COMPILE_OPTION "-std=gnu++26")
  set(CMAKE_CXX_STANDARD 26)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(GRPCXX_TICKET06_TESTS "Build compression and memory-BIO tests" ON)
  option(GRPCXX_TICKET06_BENCHMARKS "Build compression benchmarks" ON)
  find_package(OpenSSL REQUIRED)
  find_package(ZLIB REQUIRED)
  include(FetchContent)
  find_package(fmt CONFIG QUIET)
  if(NOT TARGET fmt::fmt)
    FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.1.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(fmt)
  endif()
  find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
  find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
  find_library(SNAPPY_LIBRARY NAMES snappy)
  if(NOT SNAPPY_LIBRARY)
    set(SNAPPY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(SNAPPY_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(snappy GIT_REPOSITORY https://github.com/google/snappy.git GIT_TAG 1.2.2 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(snappy)
    set(SNAPPY_LIBRARY snappy)
  endif()
  file(GLOB COMPRESSION_SOURCES CONFIGURE_DEPENDS "${GRPCXX_ROOT}/src/grpcxx/compression/*.cc")
  file(GLOB TLS_SOURCES CONFIGURE_DEPENDS "${GRPCXX_ROOT}/src/grpcxx/tls/*.cc")
  set(GRPCXX_CORE_SOURCE_ROOT "${GRPCXX_ROOT}/src" CACHE PATH "Canonical core/framing source root")
  if(NOT EXISTS "${GRPCXX_CORE_SOURCE_ROOT}/grpcxx/framing/aligned_message_buf.h")
    message(FATAL_ERROR "GRPCXX_CORE_SOURCE_ROOT must contain the canonical core/framing sources")
  endif()
  set(CORE_SOURCES "${GRPCXX_CORE_SOURCE_ROOT}/grpcxx/framing/aligned_message_buf.cc")
  if(EXISTS "${GRPCXX_CORE_SOURCE_ROOT}/grpcxx/core/status.cc")
    list(APPEND CORE_SOURCES "${GRPCXX_CORE_SOURCE_ROOT}/grpcxx/core/status.cc")
  endif()
  add_library(grpcxx_ticket06 STATIC ${COMPRESSION_SOURCES} ${TLS_SOURCES} ${CORE_SOURCES})
  target_include_directories(grpcxx_ticket06 PUBLIC "${GRPCXX_ROOT}/src" "${GRPCXX_CORE_SOURCE_ROOT}")
  target_compile_options(grpcxx_ticket06 PUBLIC -freflection -fno-exceptions -Wall -Wextra -Wswitch)
  target_link_libraries(grpcxx_ticket06 PUBLIC fmt::fmt OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB ${ZSTD_LIBRARY} ${LZ4_LIBRARY} ${SNAPPY_LIBRARY})
  add_library(grpcxx::compression ALIAS grpcxx_ticket06)
  add_library(grpcxx::tls ALIAS grpcxx_ticket06)
endif()
add_executable(tls_and_compression main.cc)
target_link_libraries(tls_and_compression PRIVATE grpcxx::compression grpcxx::tls fmt::fmt)
target_compile_definitions(tls_and_compression PRIVATE GRPCXX_TLS_FIXTURE_DIR="${GRPCXX_ROOT}/tests/units/tls")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  if(GRPCXX_TICKET06_TESTS)
    enable_testing()
    find_package(GTest CONFIG QUIET)
    if(NOT TARGET GTest::gtest_main)
      set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
      FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.16.0 GIT_SHALLOW TRUE)
      FetchContent_MakeAvailable(googletest)
    endif()
    add_subdirectory("${GRPCXX_ROOT}/tests/units/compression" compression_tests)
    add_subdirectory("${GRPCXX_ROOT}/tests/units/tls" tls_tests)
  endif()
  if(GRPCXX_TICKET06_BENCHMARKS)
    find_package(benchmark CONFIG QUIET)
    if(NOT TARGET benchmark::benchmark_main)
      set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
      FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.9.2 GIT_SHALLOW TRUE)
      FetchContent_MakeAvailable(benchmark)
    endif()
    add_subdirectory("${GRPCXX_ROOT}/benches/compression" compression_benchmarks)
  endif()
endif()
