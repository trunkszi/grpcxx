cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(order_service LANGUAGES CXX)
  enable_testing()
endif()
if(NOT TARGET fmt::fmt)
  find_package(fmt CONFIG REQUIRED)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/canonical_dependencies.cmake")
order_service_load_canonical_dependencies()

if(NOT TARGET order_gateway)
  add_library(order_gateway STATIC order_gateway_service.cc order_transport.cc)
  target_include_directories(order_gateway PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(order_gateway PUBLIC grpcxx::core grpcxx::framing
    grpcxx::reflection grpcxx::h2 grpcxx::streaming)
  add_executable(order_service main.cc)
  target_link_libraries(order_service PRIVATE order_gateway fmt::fmt)
  add_test(NAME order_service_smoke COMMAND order_service)
  set_tests_properties(order_service_smoke PROPERTIES LABELS "e2e;app;order_service;smoke")
endif()

# Deferred registration sees optional suites added later by the integrating root.
function(order_service_register_existing_smokes)
  foreach(suite cpp26_reflection_unary streaming_showcase multi_codec_showcase tls_and_compression)
    if(TARGET ${suite} AND NOT TEST ${suite}_smoke)
      add_test(NAME ${suite}_smoke COMMAND $<TARGET_FILE:${suite}>)
      set_tests_properties(${suite}_smoke PROPERTIES LABELS "e2e;app;order_service;smoke")
    endif()
  endforeach()
endfunction()
get_property(order_service_smokes_scheduled GLOBAL PROPERTY ORDER_SERVICE_SMOKES_SCHEDULED)
if(NOT order_service_smokes_scheduled)
  set_property(GLOBAL PROPERTY ORDER_SERVICE_SMOKES_SCHEDULED TRUE)
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL order_service_register_existing_smokes)
endif()
