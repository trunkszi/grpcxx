function(grpcxx_host_tool variable relative)
  set(resolved "")
  if(DEFINED VCPKG_HOST_TRIPLET AND DEFINED _VCPKG_INSTALLED_DIR
      AND NOT "${VCPKG_HOST_TRIPLET}" STREQUAL "" AND NOT "${_VCPKG_INSTALLED_DIR}" STREQUAL "")
    set(candidate "${_VCPKG_INSTALLED_DIR}/${VCPKG_HOST_TRIPLET}/tools/${relative}")
    if(EXISTS "${candidate}")
      set(resolved "${candidate}")
    endif()
  endif()
  set("${variable}" "${resolved}" PARENT_SCOPE)
endfunction()

# protoc runs this plugin as a host process. Sanitized libprotoc is built with
# -fno-sanitize-recover and aborts in DynamicMessageFactory, so the plugin links
# the unsanitized host-triplet archives.
function(_grpcxx_host_link_closure out_var)
  set(queue ${ARGN})
  set(seen "")
  set(libs "")
  while(queue)
    list(POP_FRONT queue item)
    if(item STREQUAL "" OR item MATCHES "^-")
      continue()
    endif()
    if(item MATCHES "^\\$<LINK_ONLY:(.+)>$")
      set(item "${CMAKE_MATCH_1}")
    elseif(item MATCHES "^\\$<")
      continue()
    endif()
    if(item IN_LIST seen)
      continue()
    endif()
    list(APPEND seen "${item}")
    if(NOT TARGET "${item}")
      continue()
    endif()
    get_target_property(deps "${item}" INTERFACE_LINK_LIBRARIES)
    if(deps AND NOT deps MATCHES "-NOTFOUND$")
      list(APPEND queue ${deps})
    endif()
    get_target_property(loc "${item}" IMPORTED_LOCATION_RELEASE)
    if(NOT loc OR loc MATCHES "-NOTFOUND$")
      get_target_property(loc "${item}" IMPORTED_LOCATION)
    endif()
    if(NOT loc OR loc MATCHES "-NOTFOUND$")
      continue()
    endif()
    string(REPLACE "/${VCPKG_TARGET_TRIPLET}/" "/${VCPKG_HOST_TRIPLET}/" loc "${loc}")
    if(NOT EXISTS "${loc}")
      message(FATAL_ERROR "host library missing: ${loc}")
    endif()
    list(APPEND libs "${loc}")
  endwhile()
  set("${out_var}" "${libs}" PARENT_SCOPE)
endfunction()

set(_grpcxx_host_plugin FALSE)
if(DEFINED VCPKG_HOST_TRIPLET AND DEFINED VCPKG_TARGET_TRIPLET AND DEFINED _VCPKG_INSTALLED_DIR
    AND NOT "${VCPKG_HOST_TRIPLET}" STREQUAL "${VCPKG_TARGET_TRIPLET}"
    AND "${CMAKE_CXX_FLAGS}" MATCHES "-fsanitize="
    AND EXISTS "${_VCPKG_INSTALLED_DIR}/${VCPKG_HOST_TRIPLET}/lib/libprotoc.a")
  set(_grpcxx_host_plugin TRUE)

  protoc_gen_grpcxx/main.cc
  protoc_gen_grpcxx/contract_generator.cc)

  _grpcxx_host_link_closure(_grpcxx_plugin_libs protobuf::libprotoc protobuf::libprotobuf)
  target_include_directories(protoc-gen-grpcxx BEFORE PRIVATE
    "${_VCPKG_INSTALLED_DIR}/${VCPKG_HOST_TRIPLET}/include")
  target_compile_definitions(protoc-gen-grpcxx PRIVATE FMT_HEADER_ONLY)
  target_compile_options(protoc-gen-grpcxx PRIVATE -fno-sanitize=all)
  target_link_options(protoc-gen-grpcxx PRIVATE -fno-sanitize=all)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    if(DEFINED ENV{TOOLCHAIN_ROOT} AND NOT "$ENV{TOOLCHAIN_ROOT}" STREQUAL "")
      set(_grpcxx_gcc_toolchain "$ENV{TOOLCHAIN_ROOT}/gcc-glibc")
    else()
      set(_grpcxx_gcc_toolchain "$ENV{HOME}/.local/toolchain/gcc-glibc")
    endif()
    target_compile_options(protoc-gen-grpcxx PRIVATE "--gcc-toolchain=${_grpcxx_gcc_toolchain}" -stdlib=libstdc++)
    target_link_options(protoc-gen-grpcxx PRIVATE "--gcc-toolchain=${_grpcxx_gcc_toolchain}" -stdlib=libstdc++)
  endif()
  target_link_libraries(protoc-gen-grpcxx PRIVATE
    -Wl,--start-group ${_grpcxx_plugin_libs} -Wl,--end-group pthread)
else()
  target_link_libraries(protoc-gen-grpcxx PRIVATE protobuf::libprotoc protobuf::libprotobuf fmt::fmt)
endif()

  target_link_options(protoc-gen-grpcxx PRIVATE -static)
endif()

set(grpcxx_codegen_module [=[
function(grpcxx_generate_contracts target)
  cmake_parse_arguments(GEN "" "OUTPUT_DIRECTORY;PROTOC" "FILES;IMPORT_DIRECTORIES" ${ARGN})
  if(NOT GEN_FILES OR NOT GEN_OUTPUT_DIRECTORY)
    message(FATAL_ERROR "grpcxx_generate_contracts requires FILES and OUTPUT_DIRECTORY")
  endif()
  get_filename_component(GEN_OUTPUT_DIRECTORY "${GEN_OUTPUT_DIRECTORY}" ABSOLUTE
    BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}")
  set(arguments --out "${GEN_OUTPUT_DIRECTORY}")
  foreach(directory IN LISTS GEN_IMPORT_DIRECTORIES)
    list(APPEND arguments -I "${directory}")
  endforeach()
  set(_grpcxx_host_tools "")
  if(DEFINED VCPKG_HOST_TRIPLET AND DEFINED _VCPKG_INSTALLED_DIR
      AND NOT "${VCPKG_HOST_TRIPLET}" STREQUAL "" AND NOT "${_VCPKG_INSTALLED_DIR}" STREQUAL "")
    set(_grpcxx_host_tools "${_VCPKG_INSTALLED_DIR}/${VCPKG_HOST_TRIPLET}/tools")
  endif()
  if(NOT GEN_PROTOC AND _grpcxx_host_tools AND EXISTS "${_grpcxx_host_tools}/protobuf/protoc")
    set(GEN_PROTOC "${_grpcxx_host_tools}/protobuf/protoc")
  endif()
  if(GEN_PROTOC)
    list(APPEND arguments --protoc "${GEN_PROTOC}")
  endif()
  set(_grpcxx_host_plugin "")
  if(_grpcxx_host_tools AND EXISTS "${_grpcxx_host_tools}/grpcxx/protoc-gen-grpcxx")
    set(_grpcxx_host_plugin "${_grpcxx_host_tools}/grpcxx/protoc-gen-grpcxx")
  endif()
  set(headers)
  foreach(schema IN LISTS GEN_FILES)
    get_filename_component(stem "${schema}" NAME_WLE)
    if("${GEN_OUTPUT_DIRECTORY}/${stem}.grpcxx.h" IN_LIST headers)
      message(FATAL_ERROR "Contract output name collision for ${schema}")
    endif()
    list(APPEND headers "${GEN_OUTPUT_DIRECTORY}/${stem}.grpcxx.h")
  endforeach()
  if(_grpcxx_host_plugin)
    set(plugin "${_grpcxx_host_plugin}")
  elseif(TARGET protoc-gen-grpcxx)
    set(plugin protoc-gen-grpcxx)
  elseif(TARGET grpcxx::protoc-gen-grpcxx)
    set(plugin grpcxx::protoc-gen-grpcxx)
  else()
    find_program(_grpcxx_plugin protoc-gen-grpcxx REQUIRED)
    set(plugin "${_grpcxx_plugin}")
  endif()
  if(TARGET ${plugin})
    set(plugin_command "$<TARGET_FILE:${plugin}>")
  else()
    set(plugin_command "${plugin}")
  endif()
  if(TARGET grpcxx-codegen)
    set(driver grpcxx-codegen)
  elseif(TARGET grpcxx::grpcxx-codegen)
    set(driver grpcxx::grpcxx-codegen)
  else()
    set(driver "")
  endif()
  if(driver)
    set(command "$<TARGET_FILE:${driver}>" ${arguments} ${GEN_FILES})
    if(TARGET grpcxx-codegen)
      list(PREPEND command "${CMAKE_COMMAND}" -E env
        "GRPCXX_FDL_ADAPTER=${PROJECT_SOURCE_DIR}/src/grpcxx/tools/grpcxx_codegen/fdl_contract.py")
    endif()
    set(dependencies ${driver} ${plugin})
  else()
    if(NOT GEN_PROTOC)
      set(_grpcxx_host_protoc "")
      if(_grpcxx_host_tools)
        set(_grpcxx_host_protoc "${_grpcxx_host_tools}/protobuf/protoc")
      endif()
      if(_grpcxx_host_protoc AND EXISTS "${_grpcxx_host_protoc}")
        set(GEN_PROTOC "${_grpcxx_host_protoc}")
      elseif(TARGET protobuf::protoc)
        set(GEN_PROTOC "$<TARGET_FILE:protobuf::protoc>")
      else()
        find_program(GEN_PROTOC protoc REQUIRED)
      endif()
    endif()
    set(command "${GEN_PROTOC}" "--plugin=protoc-gen-grpcxx=${plugin_command}"
      "--grpcxx_out=${GEN_OUTPUT_DIRECTORY}")
    foreach(directory IN LISTS GEN_IMPORT_DIRECTORIES)
      list(APPEND command -I "${directory}")
    endforeach()
    list(APPEND command ${GEN_FILES})
    set(dependencies ${plugin})
  endif()
  file(MAKE_DIRECTORY "${GEN_OUTPUT_DIRECTORY}")
  add_custom_command(OUTPUT ${headers}
    COMMAND ${command}
    DEPENDS ${dependencies} ${GEN_FILES}
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    VERBATIM COMMAND_EXPAND_LISTS)
  add_custom_target(${target} DEPENDS ${headers})
endfunction()

]=])
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/GrpcxxCodegen.cmake" "${grpcxx_codegen_module}")
