add_library(grpcxx_codec INTERFACE)
add_library(grpcxx::codec ALIAS grpcxx_codec)
target_include_directories(grpcxx_codec INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_link_libraries(grpcxx_codec INTERFACE grpcxx::core grpcxx::framing)

if(GRPCXX_ENABLE_PROTOBUF)
  find_package(Protobuf REQUIRED)
  add_library(grpcxx_codec_protobuf protobuf_codec.cc)
  add_library(grpcxx::codec_protobuf ALIAS grpcxx_codec_protobuf)
  target_compile_definitions(grpcxx_codec_protobuf PUBLIC GRPCXX_ENABLE_PROTOBUF)
  target_link_libraries(grpcxx_codec_protobuf PUBLIC grpcxx::codec protobuf::libprotobuf fmt::fmt)
endif()

if(GRPCXX_ENABLE_FORY)
  if(NOT TARGET fory::serialization)
    find_package(fory CONFIG QUIET)
    if(NOT TARGET fory::serialization)
      find_package(Fory CONFIG REQUIRED)
    endif()
  endif()
  add_library(grpcxx_codec_fory fory_codec.cc)
  add_library(grpcxx::codec_fory ALIAS grpcxx_codec_fory)
  target_compile_definitions(grpcxx_codec_fory PUBLIC GRPCXX_ENABLE_FORY)
  target_link_libraries(grpcxx_codec_fory PUBLIC grpcxx::codec fory::serialization fmt::fmt)
endif()

if(GRPCXX_ENABLE_CAPNP)

  add_library(grpcxx::codec_capnp ALIAS grpcxx_codec_capnp)
  target_compile_definitions(grpcxx_codec_capnp PUBLIC GRPCXX_ENABLE_CAPNP)
  target_link_libraries(grpcxx_codec_capnp PUBLIC grpcxx::codec CapnProto::capnp CapnProto::kj fmt::fmt)
endif()
