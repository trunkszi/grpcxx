cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(grpcxx_h2 LANGUAGES CXX)
endif()
find_package(fmt CONFIG REQUIRED)
find_path(GRPCXX_NGHTTP2_INCLUDE_DIR nghttp2/nghttp2.h REQUIRED)
find_library(GRPCXX_NGHTTP2_LIBRARY NAMES nghttp2 REQUIRED)
add_library(grpcxx_h2 STATIC nghttp2_server_session.cc nghttp2_client_session.cc)
add_library(grpcxx::h2 ALIAS grpcxx_h2)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.30)
  target_compile_features(grpcxx_h2 PUBLIC cxx_std_26)
else()
  target_compile_options(grpcxx_h2 PUBLIC -std=c++26)
endif()
set_target_properties(grpcxx_h2 PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(grpcxx_h2 PRIVATE -freflection)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  target_compile_options(grpcxx_h2 PRIVATE -freflection-latest)
endif()
target_compile_options(grpcxx_h2 PRIVATE -fno-exceptions -Wall -Wextra -Wswitch)
target_include_directories(grpcxx_h2 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../..")
target_include_directories(grpcxx_h2 PRIVATE "${GRPCXX_NGHTTP2_INCLUDE_DIR}")
target_link_libraries(grpcxx_h2 PUBLIC fmt::fmt PRIVATE "${GRPCXX_NGHTTP2_LIBRARY}")

if(TARGET grpcxx::core)
  target_link_libraries(grpcxx_h2 PUBLIC grpcxx::core)
endif()
